-- Maybe Int atgriež skaitli ja veselu skaitļu saraksts nav tukšs
findMaximum :: [Int] -> Maybe Int

-- Atgriež Nothing ja skaitļu saraksts ir tukšs
findMaximum [] = Nothing

-- Ja saraksts nav tukšs tiek izveidots dilstoši sakārtots saraksts
-- Tiek atgriezts saraksta pirmais elements
findMaximum numbers = 
    case sortBy (comparing Down) numbers of
        (maximumNumber : _) -> Just maximumNumber
        [] -> Nothing

reverseWord :: String -> String

-- Funkcija [Char] saraksta elementu secības apgriešanai
reverseWord input = reverse input


-- 3. UZDEVUMS ---

-- Funkcija viena vārda apstrādei
-- Funkcija saņem String un atgriež String
capitalizeWord :: String -> String

-- Drošības pārbaude.
--Pie tukšas ievades tiek atgriezta tukša virkne nevis kļūda
capitalizeWord [] = []

-- Pattern matching sadala burtu virkni uz pirmo burtu un pārējo
-- toUpper firstChar padara pirmo burtu par lielo
capitalizeWord (firstChar : rest) = toUpper firstChar : rest


-- Funkcija visa teikuma apstrādei
-- Funkcija saņem String un atgriež String
capitalizeSentence :: String -> String

-- words sentence -> sadala teikumu vārdu sarakstā
-- map capitalizeWord -> padara katra vārda pirmo burtu par lielo
-- unwords -> savieno vārdu sarakstu ievietojot starp tiem atstarpes
capitalizeSentence sentence = 
    unwords (map capitalizeWord (words sentence))

-- a tipa mainīgais kas var atgriezt skaitli, simbolu vai vārdu
secondLast :: [a] -> Maybe a

-- Ja ir tukšs saraksts tad nav priekšpēdējā elementa
secondLast [] = Nothing

-- Ja sarakstā ir tikai viens elements tad nav priekšpēdējā elementa
secondLast [_] = Nothing

-- Ja sarakstā ir tieši divi elementi, tad pirmais no tiem ir meklētais
secondLast [x, _] = Just x

-- Rekursija. Ignorējam pirmo elementu un turpinam ar atlikušo sarakstu
secondLast (_ : xs) = secondLast xs


--- 5. UZDEVUMS ---

-- Operators !! atgriež elementa i-to elementu
(!!) :: [a] -> Int -> a

-- Ja meklējamā elementa indekss atrodas ārpus saraksta robežām
[] !! _ = error "Index out of bounds"

-- Kad sasniedzam šo nosacījumu, tad pašreizējais elements ir meklētais
(x : _)  !! 0 = x

-- Ignorē pirmo elementu un saglabā pārējo sarakstu kā xs
(_ : xs) !! i
    
    -- Papildus pārbaude ja ir apstrādāts negatīvs indekss
    | i < 0     = error "Index cannot be negative"
    
    -- Ja indekss nav nulle, samazinam indeksu par 1
    | otherwise = xs !! (i - 1)


--- 6. UZDEVUMS ---
-- FUNKCIJAS DEKLARĒŠANA --

-- Ievade ir viens saraksts.
--Rezultāts -> saraksts, kura elementi paši ir saraksti
-- gaps [1, 2, 3] -> [[2,3], [1,3], [1,2]]
gaps :: [a] -> [[a]]

-- Bāzes gadījums. Ja saraksts ir tukšs tad nav arī rezultāta
gaps [] = []

-- Rekursīvais gadījums
-- (x : xs) sadala sarakstu. [1, 2, 3] -> x = 1; xs = [2, 3]
-- gaps xs sadala sarakstu rekursīvi [2, 3] -> [[3], [2]]
-- map (x :) pievieno atpakaļ katram 1: [[3], [2]] -> [[1, 3], [1, 2]]
gaps (x : xs) = xs : map (x :) (gaps xs)

-- Definējam bināra koka datu tipu
data Tree a = Branch a (Tree a) (Tree a) | Leaf
    
    -- Koka strukrtūras izvadīšanai uz ekrāna
    deriving (Show)


-- isSearchTree funkcijas definīcija. 
-- Pārbauda vai koks atbilst binārā meklēšanas koka īpašībai

-- ?????????????????????????????????????????????
isSearchTree :: Ord a => Tree a -> Bool

-- Saknes vērtībai nav ne apakšējās, ne augšējās robežas
-- lower = Nothing; upper = Nothing
isSearchTree tree = checkTree Nothing Nothing tree

-- Pārbauda vai koks atbilst binārā koka meklēšanas koka noteikumiem
-- izmantojot pieļaujamās apakšējās un augšējās robežas
-- "Ord a" -> a tipa vērtībām jābūt salīdzināmām
checkTree :: Ord a => Maybe a -> Maybe a -> Tree a -> Bool

-- Sasniegts (Leaf) tukšs zars
-- Tad tajā nav vērtību kas varētu pārkāpt BST, atgriežam True
checkTree _ _ Leaf = True

-- Ja pašreizējais elements ir Branch
-- x     = pašreizējā mezgla vērtība
-- left  = kreisais apakškoks
-- right = labais apakškoks
-- lower = minimālā atļautā robeža
-- upper = maksimālā atļautā robeža
checkTree lower upper (Branch x left right) = 
        
        -- Pārbaudam vai pašreizējā vērtība x ir virs apakšējās robežas
        withinLowerBound &&
        
        -- Pārbaudam vai pašreizēā vērtība x nepārsniedz augšējo robežu
        withinUpperBound &&

        -- Rekursīvi pārbaudam kreiso apakškoku
        -- Pašreizējā x vērtība kļūst par jauno augšējo robežu
        -- Kreisajā apakškokā vērtībām jābūt <= x
        checkTree lower (Just x) left &&

        -- Rekursīvi pārbaudam labo apakškoku
        -- Pašreizējā x vērtība kļūst par jauno apakšējo robežu
        -- Labajā apakškokā vērtībām jābūt > x
        checkTree (Just x) upper right
    
    -- Definējam lokālas palīgvērtības
    where

        -- Nosakam vai x atbilst pašreizējai apakšējai robežai
        withinLowerBound = 
            
            -- Pārbaudam lower vērtību izmantojot pattern matching
            case lower of

                -- Apakšējās robežas nav. x automātiski iztur pārbaudi
                Nothing -> True

                -- Apakšējā robeža eksistē. x jābūt lielākam par šo robežu
                Just low -> x > low

        -- Nosakam vai x atbilst pašreizējai apakšējai robežai
        withinUpperBound = 
            
            -- Pārbaudam upper vērtību izmantojot pattern matching
            case upper of
                
                -- Apakšējās robežas nav. x automātiski iztur pārbaudi
                Nothing -> True
                
                -- Augšējā robeža eksistē. x jābūt mazākam vai vienādam ar šo robežu
                Just high -> x <= high

-- Ievietojam jaunu vērtību binārājā meklēšanas kokā
insertTree :: Ord a => a -> Tree a -> Tree a

-- Ja koks ir tukšs
-- Ievietojam jaunu zaru kas satur ievietoto vērtību
insertTree e Leaf = Branch e Leaf Leaf

-- Ja kokam jau ir zars
-- Salīdzinam ievietojamo vērtību e ar pašreizējo vērtību x
insertTree e (Branch x left right)

    -- Mazākas vai vienādas vērtības tiek ievietotas kreisajā apakškokā
    | e <= x = 
        Branch x (insertTree e left) right

    -- Lielākas vērtības tiek ievietotas labajā apakškokā
    | e > x = 
        Branch x left (insertTree e right)


-- Pārbaudam īpašību ka vērtības ievietošana derīgā BST rada citu derīgu BST
insertPreservesSearchTree :: Ord a => a -> Tree a -> Bool
insertPreservesSearchTree value tree = 
    not (isSearchTree tree) ||
    isSearchTree (insertTree value tree)


-- Kļūdaina insertTree versija testēšanai
-- Lielākas vērtības kļūdaini tiek ievietotas kreisajā apakškokā
insertTreeWrong :: Ord a => a -> Tree a -> Tree a
insertTreeWrong e Leaf = Branch e Leaf Leaf

-- Ja kokam jau ir zars
-- Salīdzinam ievietojamo vērtību e ar pašreizējo vērtību x
insertTreeWrong e (Branch x left right)
    
    -- Mazākas vai vienādas vērtības tiek ievietotas kreisajā apakškokā
    | e <= x = 
        Branch x (insertTreeWrong e left) right

    -- Kļūda:
    -- Lielāka vērtība tiek nepareizi ievietota kreisajā apakškokā
    | e > x = 
        Branch x (insertTreeWrong e left) right


-- Pārbaudsam vai kļūdainā ievietošanas funkcija saglabā BST īpšību
insertWrongPreservesSearchTree :: Ord a => a -> Tree a -> Bool
insertWrongPreservesSearchTree value tree = 
    not (isSearchTree tree) || 
    isSearchTree (insertTreeWrong value tree)

-- Definējam funkciju, kas salīdzina likuma abas puses
-- Eq b: tipa b vērtībām jāatbalsta vienādības salīdzināšana ar operatoru (==)
-- (a -> b): funkcija, kas pārveido a tipa vērtību par b
-- [[a]]: saraksts, kura elementi ir citi saraksti
-- Bool: rezultāts būs True vai False
checkMapConcatLaw :: Eq b => (a -> b) -> [[a]] -> Bool

-- Funkcija saņem pārveidošanas funkciju f un sarakstu lists
checkMapConcatLaw f lists =

    -- Aprēķinām likuma kreiso pusi
    -- concat apvieno visus iekšējos sarakstus vienā sarakstā
    -- map f pielieto funkciju f katram iegūtā saraksta elementam
    -- Operators (.) apzīmē funkciju kompozīciju:
    -- vispirms izpilda labo funkciju, pēc tam kreiso
    -- Operators (==) salīdzina abu izteiksmju rezultātus
    (map f . concat) lists ==

    -- Aprēķinām likuma labo pusi
    -- map (map f): pielieto funkciju f katram elementam katrā iekšējā sarakstā
    -- concat: apvieno pārveidotos sarakstus.
    -- Ja abu pušu rezultāti sakrīt, tiek atgriezts True
    -- Pretējā gadījumā tiek atgriezts False
    (concat . map (map f)) lists

task8 :: IO ()
task8 = do

    -- 8. UZDEVUMS --
    putStrLn ("TASK 8: Prove the map and concat law.")

    -- Izveidojam sākotnējo sarakstu, kas satur trīs sarakstus
    let numbers = [[1, 2], [3, 4], [5]]

    -- Definējam funkciju f, kas katru skaitli reizina ar 2
    let f = (*2)

    -- Aprēķinām likuma kreiso pusi
    -- 1. concat apvieno sarakstus: [1,2,3,4,5]
    -- 2. map f reizina katru elementu ar 2
    -- Sagaidāmais rezultāts: [2,4,6,8,10]
    let leftResult = (map f . concat) numbers

    -- Aprēķinām likuma labo pusi
    -- 1. map (map f) izveido [[2,4],[6,8],[10]]
    -- 2. concat apvieno pārveidotos sarakstus
    -- Sagaidāmais rezultāts: [2,4,6,8,10]
    let rightResult = (concat . map (map f)) numbers

    -- Izvadām demonstrācijas nosaukumu
    putStrLn "Task 8: Map and Concat Law"
    putStrLn ""

    -- Izvadām sākotnējo sarakstu
    putStrLn ("Original lists: " ++ show numbers)

    -- Izvadam pārveidošanas funkciju
    putStrLn "Function: multiply by 2"
    putStrLn ""

    -- Izvadām likuma kreisās puses aprēķina rezultātu
    putStrLn ("Left side result:  " ++ show leftResult)

    -- Izvadām likuma labās puses aprēķina rezultātu
    putStrLn ("Right side result: " ++ show rightResult)
    putStrLn ""

    -- Izsaucam checkMapConcatLaw ar funkciju f un numbers
    -- Ja rezultāti sakrīt, terminālī parādās True
    putStrLn ("Are both sides equal? "
        ++ show (checkMapConcatLaw f numbers))
    putStrLn ""

    -- Izvadām papildu testu sadaļas nosaukumu
    putStrLn "Additional Tests"

    -- 1. tests:
    -- Pārbaudām likumu ar veseliem skaitļiem
    -- Sagaidāmais rezultāts: True
    putStrLn ("Test 1 - Multiply by 2: "
        ++ show (checkMapConcatLaw (*2) [[1,2],[3,4]]))

    -- 2. tests:
    -- Pārbaudām likumu, katram skaitlim pieskaitot 10
    -- Papildus pārbaudām tukšu iekšējo sarakstu []
    -- Sagaidāmais rezultāts: True
    putStrLn ("Test 2 - Add 10: "
        ++ show (checkMapConcatLaw (+10) [[1],[2,3],[]]))

    -- 3. tests:
    -- Pārbaudām likumu ar teksta virknēm
    -- length aprēķina katras teksta virknes garumu
    -- "Hi" tiek pārveidots par 2
    -- Sagaidāmais rezultāts: True
    putStrLn ("Test 3 - String lengths: "
        ++ show (checkMapConcatLaw length
            [["Hi","Hello"],["Haskell"]]))

    -- 4. tests:
    -- Pārbaudām likumu ar loģiskajām vērtībām
    -- not maina True uz False un False uz True
    -- Sagaidāmais rezultāts: True
    putStrLn ("Test 4 - Boolean negation: "
        ++ show (checkMapConcatLaw not
            [[True,False],[False]]))

    -- 5. tests:
    -- Pārbaudām likumu ar tukšu ārējo sarakstu
    -- ([] :: [[Int]]): tukšs saraksts, kura elementi būtu Int tipa saraksti
    -- Abas likuma puses atgriež tukšu sarakstu
    -- Sagaidāmais rezultāts: True
    putStrLn ("Test 5 - Empty list: "
        ++ show (checkMapConcatLaw (+1) ([] :: [[Int]])))
